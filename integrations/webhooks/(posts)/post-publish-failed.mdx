---
title: "post.publish_failed"
description: "Fired when a post fails to publish"
sidebar:
  label: "post.publish_failed"
---

All webhook payloads follow a consistent top-level structure with event-specific data nested within the `data` object.

<TypeTable
  type={{
    "type": { type: "string", required: true, description: "The event type (e.g. post.publish_failed)." },
    "data": { type: "object", required: true, description: "Event-specific payload for this webhook." },
    "data.post": { type: "object", required: true, description: "The post that failed to publish." },
    "data.post.id": { type: "string", required: true, description: "Post ID." },
    "data.post.title": { type: "string", required: true, description: "Post title." },
    "data.post.channel": { type: "string", required: true, description: "Channel where publish failed." },
    "data.post.error": { type: "string", required: true, description: "Error message describing the failure." },
    "data.post.campaign": { type: "object | null", description: "Campaign if assigned. Contains id, name, startDate, and endDate." },
    "data.post.url": { type: "string", required: true, description: "Ordinal post URL." },
    "data.post.profile": { type: "object", required: true, description: "Profile that attempted to publish. Contains id, name, and detail." },
    "data.post.scheduledPublishAt": { type: "string", required: true, description: "ISO 8601 timestamp when publish was attempted." },
    "data.post.workspace": { type: "object", required: true, description: "Workspace the post belongs to. Contains id, slug, and name." },
    "data.post.createdBy": { type: "object", required: true, description: "User who created the post. Contains id, firstName, lastName, and email." },
    "data.post.failedAt": { type: "string", required: true, description: "ISO 8601 timestamp when the publish failed." },
    "createdAt": { type: "string", required: true, description: "ISO 8601 timestamp when the webhook event was created." },
  }}
/>
