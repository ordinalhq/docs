---
title: "post.approval.approved"
description: "Fired when an approver grants approval for a post"
sidebar:
  label: "post.approval.approved"
---

All webhook payloads follow a consistent top-level structure with event-specific data nested within the `data` object.

<TypeTable
  type={{
    "type": { type: "string", required: true, description: "The event type (e.g. post.approval.approved)." },
    "data": { type: "object", required: true, description: "Event-specific payload for this webhook." },
    "data.approval": { type: "object", required: true, description: "The approval that was granted." },
    "data.approval.id": { type: "string", required: true, description: "Approval ID." },
    "data.approval.isBlocking": { type: "boolean", required: true, description: "Whether this approval blocked publishing until granted." },
    "data.approval.message": { type: "string", description: "Optional message attached to the approval request." },
    "data.approval.dueDate": { type: "string", description: "ISO 8601 due date for the approval, if set." },
    "data.approval.createdAt": { type: "string", required: true, description: "ISO 8601 timestamp when the approval was created." },
    "data.approval.status": { type: "string", required: true, description: "Always Approved for this event." },
    "data.approval.approvedAt": { type: "string", required: true, description: "ISO 8601 timestamp when the approver granted approval." },
    "data.approval.approvedBy": { type: "object", required: true, description: "User who approved. Contains id, firstName, lastName, and email." },
    "data.approval.requestedBy": { type: "object", required: true, description: "User who originally requested approval. Contains id, firstName, lastName, and email." },
    "data.approval.post": { type: "object", required: true, description: "Post that was approved. Contains id, title, and url." },
    "data.approval.campaign": { type: "object", description: "Campaign the post belongs to, if any. Contains id, name, startDate, and endDate." },
    "data.approval.workspace": { type: "object", required: true, description: "Workspace the post belongs to. Contains id, slug, and name." },
    "createdAt": { type: "string", required: true, description: "ISO 8601 timestamp when the webhook event was created." },
  }}
/>
