---
title: "linkedin.monitoring.comment.edited"
description: "Fired when a comment or reply on a connected LinkedIn company page post is edited"
sidebar:
  label: "linkedin.monitoring.comment.edited"
---

All webhook payloads follow a consistent top-level structure with event-specific data nested within the `data` object.

<TypeTable
  type={{
    "type": { type: "string", required: true, description: "The event type (e.g. linkedin.monitoring.comment.edited)." },
    "data": { type: "object", required: true, description: "Event-specific payload for this webhook." },
    "data.comment": { type: "object", required: true, description: "The edited comment." },
    "data.comment.id": { type: "string", required: true, description: "Comment ID." },
    "data.comment.urn": { type: "string", required: true, description: "LinkedIn comment URN." },
    "data.comment.commentary": { type: "string | null", description: "Updated comment text." },
    "data.comment.imageUrl": { type: "string | null", description: "Image attached to the comment, if any." },
    "data.comment.commentUrl": { type: "string | null", description: "LinkedIn URL that opens the comment on the post." },
    "data.comment.publishedAt": { type: "string", required: true, description: "ISO 8601 timestamp when the comment was first published." },
    "data.comment.lastModifiedAt": { type: "string", required: true, description: "ISO 8601 timestamp from LinkedIn for this edit notification." },
    "data.comment.parentCommentId": { type: "string | null", description: "Parent comment ID when this comment is a reply. Null for a comment on the post." },
    "data.comment.isCompanyAuthor": { type: "boolean", required: true, description: "Whether the company page that owns the post authored the comment." },
    "data.comment.author": { type: "object", required: true, description: "Comment author. Contains urn, name, profileImageUrl, and type (User or Organization). name, profileImageUrl, and type can be null." },
    "data.comment.post": { type: "object", required: true, description: "Company page post that was commented on. Contains id, commentary, url (Ordinal monitoring inbox), postUrl (LinkedIn), and publishedAt." },
    "data.comment.companyPage": { type: "object", required: true, description: "Company page that owns the post. Contains id, name, and profileImageUrl." },
    "data.comment.workspace": { type: "object", required: true, description: "Workspace the company page is connected to. Contains id, slug, and name." },
    "createdAt": { type: "string", required: true, description: "ISO 8601 timestamp when the webhook event was created." },
  }}
/>
