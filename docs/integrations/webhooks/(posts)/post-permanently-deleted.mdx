---
title: "post.permanently_deleted"
description: "Fired when a post is permanently deleted"
sidebar:
  label: "post.permanently_deleted"
---

All webhook payloads follow a consistent top-level structure with event-specific data nested within the `data` object.

<TypeTable
  type={{
    "type": { type: "string", required: true, description: "The event type (e.g. post.permanently_deleted)." },
    "data": { type: "object", required: true, description: "Event-specific payload for this webhook." },
    "data.post": { type: "object", required: true, description: "The permanently deleted post." },
    "data.post.id": { type: "string", required: true, description: "Post ID." },
    "data.post.title": { type: "string", required: true, description: "Post title." },
    "data.post.status": { type: "string", required: true, description: "Post status." },
    "data.post.channels": { type: "array", required: true, description: "Channels the post was scheduled for." },
    "data.post.campaign": { type: "object | null", description: "Campaign if assigned. Contains id, name, startDate, and endDate." },
    "data.post.url": { type: "string", required: true, description: "Ordinal post URL." },
    "data.post.labels": { type: "array", required: true, description: "Labels attached to the post. Each contains id, name, color, and backgroundColor." },
    "data.post.workspace": { type: "object", required: true, description: "Workspace the post belonged to. Contains id, slug, and name." },
    "data.post.deletedBy": { type: "object | null", description: "User who deleted the post, or null if auto-deleted after trash retention. Contains id, firstName, lastName, and email." },
    "data.post.deletedAt": { type: "string", required: true, description: "ISO 8601 timestamp when the post was permanently deleted." },
    "data.post.linkedIn": { type: "object | null", description: "LinkedIn content when the post included LinkedIn. Contains profile (id, name, detail), copy, and assets (id, name, url, mimetype, size, width, height, duration)." },
    "data.post.x": { type: "object | null", description: "X/Twitter content when the post included Twitter. Contains profile (id, name, detail) and tweets (array of { copy, assets })." },
    "createdAt": { type: "string", required: true, description: "ISO 8601 timestamp when the webhook event was created." },
  }}
/>
