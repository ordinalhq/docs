---
title: "post.rescheduled"
description: "Fired when a post's scheduled time is changed"
sidebar:
  label: "post.rescheduled"
---

All webhook payloads follow a consistent top-level structure with event-specific data nested within the `data` object.

<TypeTable
  type={{
    "type": { type: "string", required: true, description: "The event type (e.g. post.rescheduled)." },
    "data": { type: "object", required: true, description: "Event-specific payload for this webhook." },
    "data.post": { type: "object", required: true, description: "The rescheduled post." },
    "data.post.id": { type: "string", required: true, description: "Post ID." },
    "data.post.title": { type: "string", required: true, description: "Post title." },
    "data.post.channels": { type: "array", required: true, description: "Channels the post is scheduled for." },
    "data.post.campaign": { type: "object | null", description: "Campaign if assigned. Contains id, name, startDate, and endDate." },
    "data.post.url": { type: "string", required: true, description: "Ordinal post URL." },
    "data.post.labels": { type: "array", required: true, description: "Labels attached to the post. Each contains id, name, color, and backgroundColor." },
    "data.post.previousPublishAt": { type: "string", required: true, description: "ISO 8601 timestamp of the previous publish time." },
    "data.post.newPublishAt": { type: "string", required: true, description: "ISO 8601 timestamp of the new publish time." },
    "data.post.workspace": { type: "object", required: true, description: "Workspace the post belongs to. Contains id, slug, and name." },
    "data.post.rescheduledBy": { type: "object", required: true, description: "User who rescheduled the post. Contains id, firstName, lastName, and email." },
    "data.post.rescheduledAt": { type: "string", required: true, description: "ISO 8601 timestamp when the post was rescheduled." },
    "data.post.linkedIn": { type: "object | null", description: "LinkedIn content when the post includes LinkedIn. Contains profile (id, name, detail), copy, and assets (id, name, url, mimetype, size, width, height, duration)." },
    "data.post.x": { type: "object | null", description: "X/Twitter content when the post includes Twitter. Contains profile (id, name, detail) and tweets (array of { copy, assets })." },
    "createdAt": { type: "string", required: true, description: "ISO 8601 timestamp when the webhook event was created." },
  }}
/>
